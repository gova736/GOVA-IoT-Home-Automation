#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <FastLED.h>
#include <DHT.h>

#define STRIP_PIN D4
#define NUM_LEDS 7

#define SINGLE_LED D1

#define DHT_PIN D2
#define DHT_TYPE DHT11

CRGB leds[NUM_LEDS];

DHT dht(DHT_PIN, DHT_TYPE);

ESP8266WebServer server(80);

int brightnessValue = 50;
float temperatureLimit = 30.0;

bool autoMode = true;
bool rainbowMode = false;
bool ledState = false;

byte rainbowHue = 0;


// ---------- RGB COLOR ----------

void setColor(CRGB color)
{
  rainbowMode = false;

  FastLED.clear();
  fill_solid(leds, NUM_LEDS, color);

  FastLED.setBrightness(brightnessValue);
  FastLED.show();
}

// ---------- RAINBOW ----------

void updateRainbow()
{
  if (!rainbowMode)
    return;

  static unsigned long lastRainbow = 0;

  if (millis() - lastRainbow < 5)
    return;

  lastRainbow = millis();

  for (int i = 0; i < NUM_LEDS; i++)
  {
    leds[i] = CHSV(rainbowHue + (i * 30), 255, 255);
  }

  FastLED.setBrightness(brightnessValue);
  FastLED.show();

  rainbowHue++;
}
// ---------- TEMPERATURE AUTO LED ----------

// ---------- TEMPERATURE AUTO LED ----------

void checkTemperature()
{
  if (!autoMode)
    return;

  float temperature = dht.readTemperature();

  if (isnan(temperature))
    return;

  if (temperature >= temperatureLimit)
  {
    digitalWrite(SINGLE_LED, LOW);
    ledState = true;
  }
  else
  {
    digitalWrite(SINGLE_LED, HIGH);
    ledState = false;
  }
}

// =================================================
// SETUP
// =================================================

void setup()
{
  pinMode(SINGLE_LED, OUTPUT);

  digitalWrite(SINGLE_LED, LOW);

  FastLED.addLeds<WS2812B, STRIP_PIN, GRB>(
    leds,
    NUM_LEDS
  );

  FastLED.setBrightness(brightnessValue);

  FastLED.clear();
  FastLED.show();

  dht.begin();

  WiFi.mode(WIFI_AP);

  WiFi.softAP(
    "ESP8266_LED",
    "12345678"
  );


  // MAIN PAGE

  server.on("/", []()
  {
    server.send(200, "text/html", webpage());
  });


  // COLORS

server.on("/red", []()
{
  setColor(CRGB(255, 0, 0));

  server.sendHeader("Location", "/");
  server.send(303);
});
  server.on("/green", []()
{
  setColor(CRGB(0, 255, 0));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/blue", []()
{
  setColor(CRGB(0, 0, 255));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/white", []()
{
  setColor(CRGB(255, 255, 255));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/purple", []()
{
  setColor(CRGB(128, 0, 255));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/yellow", []()
{
  setColor(CRGB(255, 255, 0));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/cyan", []()
{
  setColor(CRGB(0, 255, 255));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/pink", []()
{
  setColor(CRGB(255, 20, 147));
  server.sendHeader("Location", "/");
  server.send(303);
});

server.on("/orange", []()
{
  setColor(CRGB(255, 100, 0));
  server.sendHeader("Location", "/");
  server.send(303);
});

  // RAINBOW

  server.on("/rainbow", []()
{
  rainbowMode = true;

  server.sendHeader("Location", "/");
  server.send(303);
});


  // STRIP OFF

  server.on("/stripoff", []()
{
  rainbowMode = false;

  fill_solid(leds, NUM_LEDS, CRGB::Black);
  FastLED.show();

  server.sendHeader("Location", "/");
  server.send(303);
});

  // BRIGHTNESS

  server.on("/brightness", []()
  {
    if (server.hasArg("value"))
    {
      brightnessValue = server.arg("value").toInt();

      if (brightnessValue < 0)
        brightnessValue = 0;

      if (brightnessValue > 255)
        brightnessValue = 255;

      FastLED.setBrightness(brightnessValue);
      FastLED.show();
    }

    server.send(200, "text/html", webpage());
  });


  // TEMPERATURE LIMIT

  server.on("/limit", []()
  {
    if (server.hasArg("value"))
    {
      temperatureLimit = server.arg("value").toFloat();

      if (temperatureLimit < 20)
        temperatureLimit = 20;

      if (temperatureLimit > 50)
        temperatureLimit = 50;
    }

    server.send(200, "text/html", webpage());
  });


  // LED ON

  server.on("/ledon", []()
  {
    autoMode = false;

    digitalWrite(SINGLE_LED, HIGH);

    ledState = true;

    server.send(200, "text/html", webpage());
  });


  // LED OFF

  server.on("/ledoff", []()
  {
    autoMode = false;

    digitalWrite(SINGLE_LED, LOW);

    ledState = false;

    server.send(200, "text/html", webpage());
  });


  // AUTO ON

  server.on("/autoon", []()
  {
    autoMode = true;

    server.send(200, "text/html", webpage());
  });


  // AUTO OFF

  server.on("/autooff", []()
  {
    autoMode = false;

    digitalWrite(SINGLE_LED, LOW);

    ledState = false;

    server.send(200, "text/html", webpage());
  });


  server.begin();
}


// =================================================
// LOOP
// =================================================

void loop()
{
  server.handleClient();

  static unsigned long lastDHT = 0;

  if (millis() - lastDHT >= 2000)
  {
    lastDHT = millis();
    checkTemperature();
  }

  updateRainbow();

  delay(2);
}
